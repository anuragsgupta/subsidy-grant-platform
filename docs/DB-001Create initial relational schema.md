-- 1. APP_USER
-- Stores application user profiles.
-- auth_user_id can be linked to Supabase Auth later.

CREATE TABLE IF NOT EXISTS public.app_user (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    auth_user_id UUID UNIQUE
        REFERENCES auth.users(id) ON DELETE SET NULL,

    full_name VARCHAR(150) NOT NULL,
    email VARCHAR(255) NOT NULL UNIQUE,
    phone VARCHAR(20),

    active BOOLEAN NOT NULL DEFAULT TRUE,

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);


-- 2. ROLE
-- Defines the roles available in the platform.

CREATE TABLE IF NOT EXISTS public.role (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    name VARCHAR(50) NOT NULL UNIQUE,
    description VARCHAR(255),

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);


-- 3. USER_ROLE
-- Maps users to roles (many-to-many relationship).

CREATE TABLE IF NOT EXISTS public.user_role (
    user_id UUID NOT NULL
        REFERENCES public.app_user(id) ON DELETE CASCADE,

    role_id BIGINT NOT NULL
        REFERENCES public.role(id) ON DELETE RESTRICT,

    assigned_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    PRIMARY KEY (user_id, role_id)
);


-- 4. SCHEME
-- Stores government subsidy and grant schemes.

CREATE TABLE IF NOT EXISTS public.scheme (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    name VARCHAR(200) NOT NULL,
    code VARCHAR(50) NOT NULL UNIQUE,
    description TEXT,

    start_date DATE NOT NULL,
    end_date DATE NOT NULL,

    min_grant_amount NUMERIC(15,2) NOT NULL DEFAULT 0,
    max_grant_amount NUMERIC(15,2) NOT NULL,

    status VARCHAR(20) NOT NULL DEFAULT 'DRAFT'
        CHECK (status IN ('DRAFT', 'OPEN', 'PAUSED', 'CLOSED')),

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT chk_scheme_dates
        CHECK (end_date >= start_date),

    CONSTRAINT chk_scheme_grant_amount
        CHECK (
            min_grant_amount >= 0
            AND max_grant_amount >= min_grant_amount
        )
);


-- 5. BENEFICIARY
-- Stores applicant profile information.
-- A beneficiary may be linked to an application user.

CREATE TABLE IF NOT EXISTS public.beneficiary (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    user_id UUID UNIQUE
        REFERENCES public.app_user(id) ON DELETE SET NULL,

    beneficiary_type VARCHAR(30) NOT NULL
        CHECK (
            beneficiary_type IN (
                'INDIVIDUAL',
                'ORGANIZATION',
                'BUSINESS'
            )
        ),

    full_name VARCHAR(150) NOT NULL,
    identity_reference VARCHAR(100) UNIQUE,

    email VARCHAR(255),
    phone VARCHAR(20),

    address TEXT,
    region_code VARCHAR(30),

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);


-- 6. APPLICATION
-- Stores grant applications submitted by beneficiaries.

CREATE TABLE IF NOT EXISTS public.application (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    reference_no VARCHAR(40) NOT NULL UNIQUE,

    beneficiary_id UUID NOT NULL
        REFERENCES public.beneficiary(id) ON DELETE RESTRICT,

    scheme_id UUID NOT NULL
        REFERENCES public.scheme(id) ON DELETE RESTRICT,

    requested_amount NUMERIC(15,2) NOT NULL,

    status VARCHAR(30) NOT NULL DEFAULT 'DRAFT'
        CHECK (
            status IN (
                'DRAFT',
                'SUBMITTED',
                'UNDER_REVIEW',
                'CLARIFICATION_REQUIRED',
                'FIELD_VERIFIED',
                'DISTRICT_APPROVED',
                'FINANCE_APPROVED',
                'REJECTED',
                'DISBURSEMENT_IN_PROGRESS',
                'COMPLETED',
                'ON_HOLD'
            )
        ),

    submitted_at TIMESTAMPTZ,

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT chk_application_amount
        CHECK (requested_amount > 0)
);


-- =========================================================
-- TIMESTAMP UPDATE FUNCTION
-- Automatically updates updated_at on record modification.
-- =========================================================

CREATE OR REPLACE FUNCTION public.update_updated_at()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$;


-- Apply timestamp triggers.

DROP TRIGGER IF EXISTS trg_app_user_updated_at
ON public.app_user;

CREATE TRIGGER trg_app_user_updated_at
BEFORE UPDATE ON public.app_user
FOR EACH ROW
EXECUTE FUNCTION public.update_updated_at();


DROP TRIGGER IF EXISTS trg_scheme_updated_at
ON public.scheme;

CREATE TRIGGER trg_scheme_updated_at
BEFORE UPDATE ON public.scheme
FOR EACH ROW
EXECUTE FUNCTION public.update_updated_at();


DROP TRIGGER IF EXISTS trg_beneficiary_updated_at
ON public.beneficiary;

CREATE TRIGGER trg_beneficiary_updated_at
BEFORE UPDATE ON public.beneficiary
FOR EACH ROW
EXECUTE FUNCTION public.update_updated_at();


DROP TRIGGER IF EXISTS trg_application_updated_at
ON public.application;

CREATE TRIGGER trg_application_updated_at
BEFORE UPDATE ON public.application
FOR EACH ROW
EXECUTE FUNCTION public.update_updated_at();

SELECT
    a.reference_no,
    b.full_name AS beneficiary_name,
    s.name AS scheme_name,
    a.requested_amount,
    a.status
FROM public.application a
JOIN public.beneficiary b
    ON a.beneficiary_id = b.id
JOIN public.scheme s
    ON a.scheme_id = s.id
ORDER BY a.created_at;
